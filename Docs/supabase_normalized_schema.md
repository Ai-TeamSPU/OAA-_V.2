# 🗄️ การออกแบบโครงสร้างฐานข้อมูลแบบแยกตาราง (Database Normalization) บน Supabase

> **ระบบ:** OAA โปรแกรมบันทึกอาจารย์พิเศษ — มหาวิทยาลัยศรีปทุม  
> **ชนิดข้อมูล ID:** ตัวเลขเรียงลำดับอัตโนมัติ (Auto-Increment Number ID: `1`, `2`, `3`...)

---

## 💡 1. การเปลี่ยนชนิดข้อมูล ID เป็นตัวเลข (Auto-Increment Number ID)

ทำได้แน่นอนครับ! ใน PostgreSQL/Supabase เราจะใช้ชนิดข้อมูล **`BIGINT GENERATED BY DEFAULT AS IDENTITY`** (หรือ `BIGSERIAL`) แทน UUID 

### ข้อดีของ ID แบบตัวเลข:
1. **ตัวเลขอ่านง่าย สั้น กระชับ:** เช่น ID = `1`, `2`, `3` แทนรหัสยาวแบบ UUID (`550e8400-e29b-41d4...`)
2. **รันลำดับให้อัตโนมัติ:** เมื่อบันทึกข้อมูลใหม่ Supabase จะรันตัวเลขถัดไปให้อัตโนมัติโดยไม่ซ้ำกัน
3. **ประสิทธิภาพการเรียงลำดับ:** สามารถเรียงตาม `id DESC` เพื่อดึงรายการล่าสุดขึ้นก่อนได้ทันที

---

## 📐 2. แผนผังโครงสร้างตาราง (Relational Schema Design)

```
┌────────────────────────────────┐
│      1. instructors (หลัก)     │
├────────────────────────────────┤
│ - id (BIGINT - PK)             │ ← ตัวเลข 1, 2, 3...
│ - semester, faculty, branch    │
│ - first_name_th, last_name_th  │
│ - phone, email, address        │
│ - pdpa_consent, pdpa_consent_at│
└───────────────┬────────────────┘
                │
     ┌──────────┼────────────────┬────────────────┐
     │ 1:M      │ 1:M            │ 1:M            │ 1:M
     ▼          ▼                ▼                ▼
┌───────────┐ ┌──────────────┐ ┌───────────┐ ┌──────────────┐
│ 2. edu... │ │ 3. exp...    │ │ 4. awards │ │ 5. courses   │
├───────────┤ ├──────────────┤ ├───────────┤ ├──────────────┤
│ - id (PK) │ │ - id (PK)    │ │ - id (PK) │ │ - id (PK)    │
│ - inst_id │ │ - inst_id    │ │ - inst_id │ │ - inst_id    │
│ (BIGINT)  │ │ (BIGINT)     │ │ (BIGINT)  │ │ (BIGINT)     │
└───────────┘ └──────────────┘ └───────────┘ └──────────────┘
```

## 📜 3. โค้ด SQL ฉบับปรับปรุง (ใช้ ID ตัวเลขรันอัตโนมัติ)

คัดลอกโค้ด SQL ด้านล่างนี้ไปรันใน **Supabase Dashboard → SQL Editor → New Query → Run** ได้เลยครับ:

```sql
-- ========================================================
-- 1. ตารางหลัก: instructors (ข้อมูลทั่วไปและข้อมูลส่วนบุคคล)
-- ========================================================
CREATE TABLE IF NOT EXISTS public.instructors (
    -- ใช้ตัวเลขรันอัตโนมัติ (1, 2, 3, 4...)
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    
    -- ข้อที่ 1: ข้อมูลทั่วไป
    semester VARCHAR(50) NOT NULL,
    faculty VARCHAR(250) NOT NULL,
    branch VARCHAR(250) NOT NULL,
    
    -- ข้อที่ 2: ข้อมูลส่วนตัว
    title_prefix VARCHAR(50),
    title_custom VARCHAR(100),
    first_name_th VARCHAR(150) NOT NULL,
    last_name_th VARCHAR(150) NOT NULL,
    first_name_en VARCHAR(150),
    last_name_en VARCHAR(150),
    phone VARCHAR(50) NOT NULL,
    email VARCHAR(150),
    address TEXT NOT NULL,
    
    -- ข้อที่ 6: คุณสมบัติและความเชี่ยวชาญ
    teaching_proportion VARCHAR(50),
    teaching_hours VARCHAR(50),
    teaching_note TEXT,
    expertise TEXT,
    note TEXT,
    qual_subs JSONB DEFAULT '{}'::jsonb,
    qual_fields JSONB DEFAULT '{}'::jsonb,
    
    -- ข้อที่ 7: การยินยอม PDPA
    pdpa_consent BOOLEAN NOT NULL DEFAULT false,
    pdpa_consent_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
    
    -- เวลาสร้างและแก้ไขล่าสุด
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- ========================================================
-- 2. ตารางย่อย: instructor_educations (ประวัติการศึกษา)
-- ========================================================
CREATE TABLE IF NOT EXISTS public.instructor_educations (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    instructor_id BIGINT NOT NULL REFERENCES public.instructors(id) ON DELETE CASCADE,
    
    level VARCHAR(100),         -- ระดับการศึกษา (ป.ตรี/โท/เอก)
    curriculum VARCHAR(250),    -- ชื่อหลักสูตร/วุฒิ
    major VARCHAR(250),         -- สาขาวิชา
    institution VARCHAR(250),   -- สถาบันการศึกษา
    graduation_year VARCHAR(10),-- ปี พ.ศ. ที่สำเร็จ
    sequence_no INT DEFAULT 1   -- ลำดับการเรียง
);

-- ========================================================
-- 3. ตารางย่อย: instructor_experiences (ประวัติการทำงาน)
-- ========================================================
CREATE TABLE IF NOT EXISTS public.instructor_experiences (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    instructor_id BIGINT NOT NULL REFERENCES public.instructors(id) ON DELETE CASCADE,
    
    position VARCHAR(250),     -- ตำแหน่งงาน
    company VARCHAR(250),      -- หน่วยงาน/บริษัท
    start_date VARCHAR(50),    -- วันที่เริ่ม
    end_date VARCHAR(50),      -- วันที่สิ้นสุด
    is_current BOOLEAN DEFAULT false, -- ปัจจุบันยังทำอยู่หรือไม่
    is_direct BOOLEAN DEFAULT false,  -- ประสบการณ์ตรงสาขาที่สอนหรือไม่
    sequence_no INT DEFAULT 1
);

-- ========================================================
-- 4. ตารางย่อย: instructor_awards (รางวัลและผลงาน)
-- ========================================================
CREATE TABLE IF NOT EXISTS public.instructor_awards (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    instructor_id BIGINT NOT NULL REFERENCES public.instructors(id) ON DELETE CASCADE,
    
    title TEXT NOT NULL,       -- ชื่อรางวัล/ผลงาน
    link TEXT,                 -- ลิงก์แนบหลักฐาน
    sequence_no INT DEFAULT 1
);

-- ========================================================
-- 5. ตารางย่อย: instructor_courses (รายวิชาที่สอน)
-- ========================================================
CREATE TABLE IF NOT EXISTS public.instructor_courses (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    instructor_id BIGINT NOT NULL REFERENCES public.instructors(id) ON DELETE CASCADE,
    
    subject VARCHAR(250),      -- รหัสและชื่อวิชา
    credits VARCHAR(50),       -- หน่วยกิต
    teach_count VARCHAR(50),   -- กลุ่มเรียน/จำนวนกลุ่ม
    proportion VARCHAR(50),    -- สัดส่วนการสอน (%)
    degree_level VARCHAR(100), -- ระดับปริญญาที่เปิดสอน
    sequence_no INT DEFAULT 1
);

-- ========================================================
-- 6. การสร้าง Indexes เพื่อเพิ่มความเร็วในการค้นหา
-- ========================================================
CREATE INDEX IF NOT EXISTS idx_instructors_faculty ON public.instructors(faculty);
CREATE INDEX IF NOT EXISTS idx_instructors_semester ON public.instructors(semester);
CREATE INDEX IF NOT EXISTS idx_instructors_name ON public.instructors(first_name_th, last_name_th);

CREATE INDEX IF NOT EXISTS idx_edu_inst_id ON public.instructor_educations(instructor_id);
CREATE INDEX IF NOT EXISTS idx_exp_inst_id ON public.instructor_experiences(instructor_id);
CREATE INDEX IF NOT EXISTS idx_awards_inst_id ON public.instructor_awards(instructor_id);
CREATE INDEX IF NOT EXISTS idx_courses_inst_id ON public.instructor_courses(instructor_id);

-- ========================================================
-- 7. ตั้งค่า Row Level Security (RLS) เพื่อความปลอดภัย
-- ========================================================
ALTER TABLE public.instructors ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.instructor_educations ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.instructor_experiences ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.instructor_awards ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.instructor_courses ENABLE ROW LEVEL SECURITY;

-- นโยบายเปิดให้เพิ่มข้อมูลและอ่านข้อมูลได้ตามสิทธิ์
CREATE POLICY "Allow public insert" ON public.instructors FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public select" ON public.instructors FOR SELECT USING (true);
CREATE POLICY "Allow public update" ON public.instructors FOR UPDATE USING (true);
CREATE POLICY "Allow public delete" ON public.instructors FOR DELETE USING (true);

CREATE POLICY "Allow child insert" ON public.instructor_educations FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow child select" ON public.instructor_educations FOR SELECT USING (true);

CREATE POLICY "Allow child insert" ON public.instructor_experiences FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow child select" ON public.instructor_experiences FOR SELECT USING (true);

CREATE POLICY "Allow child insert" ON public.instructor_awards FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow child select" ON public.instructor_awards FOR SELECT USING (true);

CREATE POLICY "Allow child insert" ON public.instructor_courses FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow child select" ON public.instructor_courses FOR SELECT USING (true);
```
